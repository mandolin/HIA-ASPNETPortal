        public ExpenseReimbursementReviewResult ReviewRequest(ExpenseReimbursementReviewRequest request)
        {
            try
            {
                string status = MapActionToStatus(request.ActionKey);
                int rows = context.Database.ExecuteSqlCommand(
                    "UPDATE dbo.PortalBiz_ExpenseReimbursements SET Status = @Status, ReviewedByUserId = @ReviewedByUserId, ReviewedBy = @ReviewedBy, ReviewedUtc = @ReviewedUtc, ReviewComment = @ReviewComment WHERE RequestId = @RequestId;",
                    new SqlParameter("@Status", status),
                    new SqlParameter("@ReviewedByUserId", request.ReviewedByUserId),
                    new SqlParameter("@ReviewedBy", (object)request.ReviewedBy ?? DBNull.Value),
                    new SqlParameter("@ReviewedUtc", request.ReviewedUtc),
                    new SqlParameter("@ReviewComment", (object)request.ReviewComment ?? DBNull.Value),
                    new SqlParameter("@RequestId", request.RequestId));
                if (rows <= 0)
                {
                    return new ExpenseReimbursementReviewResult { Succeeded = false, Message = "NotFound", RequestId = request.RequestId };
                }

                return new ExpenseReimbursementReviewResult { Succeeded = true, RequestId = request.RequestId };
            }
            catch
            {
                return new ExpenseReimbursementReviewResult { Succeeded = false, Message = "DbError", RequestId = request.RequestId };
            }
        }
