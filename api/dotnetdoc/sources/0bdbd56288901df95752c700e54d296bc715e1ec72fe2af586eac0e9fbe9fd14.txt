        internal PortalCapabilityDefinition(
            string capabilityId,
            string layer,
            string lifecycleState,
            string primaryModuleId,
            string permissionKeyPrefix,
            string displayNameZh,
            string displayNameEn)

            DisplayNameZh = displayNameZh ?? string.Empty;

        }
