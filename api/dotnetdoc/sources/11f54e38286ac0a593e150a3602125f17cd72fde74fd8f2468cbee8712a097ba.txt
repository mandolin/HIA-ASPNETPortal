        public static bool TryDecryptSubmittedPassword(
            HttpContext context,
            string encryptedPassword,
            out string password,
            out string failureCode,
            out string eventId)
        {

                context,
                encryptedPassword,
                out password,
                out failureCode,
                out eventId);
        }
