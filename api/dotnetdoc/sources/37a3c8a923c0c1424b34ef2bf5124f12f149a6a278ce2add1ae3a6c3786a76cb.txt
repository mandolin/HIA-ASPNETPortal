        public IContactItem GetSingleContact(int itemId)

            //   <zh-CN>编辑入口的 ItemId 来自请求，未命中返回空值，让页面层统一输出低敏提示或拒绝访问响应。</zh-CN>

            // </lang>
            return _context.Contacts.SingleOrDefault(i => i.ItemId == itemId);
