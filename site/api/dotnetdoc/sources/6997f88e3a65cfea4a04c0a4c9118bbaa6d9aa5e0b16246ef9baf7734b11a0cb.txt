    public enum PortalCookieSameSiteMode

        /// <summary>

        /// </lang>
        /// </summary>

        /// <summary>
        /// <lang>
        ///   <zh-CN>下发 SameSite=Strict。</zh-CN>

        /// </lang>
        /// </summary>

        /// <summary>

        /// </lang>
        /// </summary>

    }
