        protected override void OnModelCreating(DbModelBuilder modelBuilder)
        {

                .HasKey(item => item.OrganizationUnitId);

            modelBuilder.Entity<EmployeeItem>()
                .HasKey(item => item.EmployeeId);

                .HasKey(item => item.BindingId);
        }
