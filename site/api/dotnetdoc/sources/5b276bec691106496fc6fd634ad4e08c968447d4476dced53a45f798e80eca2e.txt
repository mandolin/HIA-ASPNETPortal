        protected void Page_Load(object sender, EventArgs e)
        {
            if (!TryInitializeRequest() ||
                !PortalAuthorization.EnsurePermission(Context, PortalPermissionKeys.AdminRolesEdit))
            {
                return;
            }

            ConfigurePasswordSubmission();

            if (!Page.IsPostBack)
            {
                BindData();
            }
        }
